# Tradibu crawl rules.
#
# ONE user-agent group on purpose. A crawler that matches a named group ignores
# the "*" group entirely, so the previous file - which gave Googlebot/Bingbot
# their own "Allow: /" groups with no Disallow lines - actually let Googlebot
# crawl /admin/, /buyer/ and /checkout/. Every rule below applies to every
# crawler now.
User-agent: *
Allow: /

# Authenticated / transactional surfaces (never useful in search results).
Disallow: /admin/
Disallow: /buyer/
Disallow: /dashboard
Disallow: /checkout
Disallow: /order-success/
Disallow: /auth/
Disallow: /wishlist/

# Seller dashboard subtrees must be listed one by one: a single
# "Disallow: /seller/" would also block the PUBLIC storefronts at
# /seller/<seller-id>, which are meant to be indexed.
Disallow: /seller/dashboard
Disallow: /seller/products
Disallow: /seller/store
Disallow: /seller/collections
Disallow: /seller/orders
Disallow: /seller/reviews
Disallow: /seller/analytics
Disallow: /seller/ads
Disallow: /seller/wallet
Disallow: /seller/chat
